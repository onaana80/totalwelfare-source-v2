// --- SHARED STATE MANAGER (Universal Logic for All Pages) --- // This script handles your TTVc Vault, Promoter Income, and Owner Income // It loads from 'localStorage' so your counts update even if you switch pages. (function() { // 1. Load State from Storage (Prevent errors on load) const savedHoldings = JSON.parse(localStorage.getItem('userHoldings')) || '{}'; const savedPromoterBalance = parseFloat(localStorage.getItem('userPromoterBalance')) || 0.00; const savedOwnerIncome = parseFloat(localStorage.getItem('userOwnerIncome')) || 0.00; // 2. Save State Helper Functions window.saveHoldings = function(holdings) { const json = JSON.stringify(holdings); localStorage.setItem('userHoldings', json); updateDashboardVisuals(); // Update any visible badges immediately }; window.savePromoterBalance = function(amount) { localStorage.setItem('userPromoterBalance', amount.toString()); updateDashboardVisuals(); }; window.saveOwnerIncome = function(amount) { localStorage.setItem('userOwnerIncome', amount.toString()); updateDashboardVisuals(); }; // 3. Update Visuals Helper (Connects HTML elements to Data) window.updateDashboardVisuals = function() { const vaultBadge = document.querySelector('.wallet-badge'); if(vaultBadge) vaultBadge.innerText = Object.values(savedHoldings).length; // Update Income Badges const promoterBadge = document.getElementById('dash-promoter'); const ownerBadge = document.getElementById('dash-owner'); if(promoterBadge) promoterBadge.innerText = `$${savedPromoterBalance.toFixed(2)}`; if(ownerBadge) ownerBadge.innerText = `$${savedOwnerIncome.toFixed(2)}`; }; // 4. Expose Helper Functions (For use in other pages) window.getSavedHoldings = function() { return savedHoldings; }; window.getSavedPromoterBalance = function() { return savedPromoterBalance; }; window.getSavedOwnerIncome = function() { return savedOwnerIncome; }; console.log("State Manager Loaded. Vault: " + Object.values(savedHoldings).length); })();
